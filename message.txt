local function updatePlayerList()
    for _, child in ipairs(ScrollingList:GetChildren()) do
        if child:IsA("TextButton") then
            child:Destroy()
        end
    end
    
    local count = 0
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= localPlayer then
            count = count + 1
            local pButton = Instance.new("TextButton")
            pButton.Size = UDim2.new(1, -10, 0, 30)
            pButton.BackgroundColor3 = Color3.fromRGB(60, 60, 60)
            pButton.TextColor3 = Color3.fromRGB(255, 255, 255)
            pButton.TextSize = 13
            pButton.Font = Enum.Font.SourceSans
            pButton.Text = player.Name
            pButton.Parent = ScrollingList
            
            local btnCorner = Instance.new("UICorner")
            btnCorner.CornerRadius = UDim.new(0, 5)
            btnCorner.Parent = pButton
            
            pButton.MouseButton1Click:Connect(function()
                if player.Character and player.Character:FindFirstChild("HumanoidRootPart") and localPlayer.Character and localPlayer.Character:FindFirstChild("HumanoidRootPart") then
                    localPlayer.Character.HumanoidRootPart.CFrame = player.Character.HumanoidRootPart.CFrame + Vector3.new(0, 3, 0)
                end
            end)
        end
    end
    ScrollingList.CanvasSize = UDim2.new(0, 0, 0, count * 35)
end

Players.PlayerAdded:Connect(updatePlayerList)
Players.PlayerRemoving:Connect(updatePlayerList)
updatePlayerList()

local function toggleSpeed()
    speedEnabled = not speedEnabled
    if speedEnabled then
        SpeedButton.Text = "السرعة (B): مفعلة (" .. customSpeed .. ")"
        SpeedButton.BackgroundColor3 = Color3.fromRGB(50, 200, 50)
    else
        SpeedButton.Text = "السرعة (B): معطلة"
        SpeedButton.BackgroundColor3 = Color3.fromRGB(200, 50, 50)
    end
end

SpeedButton.MouseButton1Click:Connect(toggleSpeed)

UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if not gameProcessed then
        if input.KeyCode == Enum.KeyCode.T then
            isAiming = not isAiming
            if isAiming then
                AimButton.Text = "التتبع (Aimlock): مفعل"
                AimButton.BackgroundColor3 = Color3.fromRGB(50, 200, 50)
            else
                AimButton.Text = "التتبع (Aimlock): معطل"
                AimButton.BackgroundColor3 = Color3.fromRGB(200, 50, 50)
            end
        elseif input.KeyCode == Enum.KeyCode.B then
            toggleSpeed()
        end
    end
end)

RunService.Stepped:Connect(function()
    if noclipEnabled and localPlayer.Character then
        for _, part in ipairs(localPlayer.Character:GetDescendants()) do
            if part:IsA("BasePart") then
                part.CanCollide = false
            end
        end
    end
end)

RunService.RenderStepped:Connect(function()
    local screenSize = camera.ViewportSize
    FOVCircle.Position = UDim2.new(0, screenSize.X / 2, 0, screenSize.Y / 2)
    FOVCircle.Size = UDim2.new(0, fovRadius * 2, 0, fovRadius * 2)

    if localPlayer.Character and localPlayer.Character:FindFirstChildOfClass("Humanoid") then
        local humanoid = localPlayer.Character:FindFirstChildOfClass("Humanoid")
        if speedEnabled then
            humanoid.WalkSpeed = customSpeed
        else
            humanoid.WalkSpeed = 16
        end
    end

    for player, objs in pairs(espObjects) do
        local character = player.Character
        if espEnabled and character and character:FindFirstChild("HumanoidRootPart") and character:FindFirstChildOfClass("Humanoid") and character.Humanoid.Health > 0 then
            local rootPart = character.HumanoidRootPart
            local head = character:FindFirstChild("Head") or rootPart
            local vector, onScreen = camera:WorldToViewportPoint(rootPart.Position)
            
            if onScreen then
                local headVector = camera:WorldToViewportPoint(head.Position + Vector3.new(0, 0.5, 0))
                local legVector = camera:WorldToViewportPoint(rootPart.Position - Vector3.new(0, 3, 0))
                local height = math.abs(headVector.Y - legVector.Y)
                local width = height / 2

                objs.Box.Size = Vector2.new(width, height)
                objs.Box.Position = Vector2.new(vector.X - width / 2, headVector.Y)
                
                if player.Team == localPlayer.Team and localPlayer.Team then
                    objs.Box.Color = Color3.fromRGB(0, 255, 0)
                else
                    objs.Box.Color = Color3.fromRGB(255, 0, 0)
                end
                objs.Box.Visible = true

                objs.Text.Text = player.Name
                objs.Text.Position = Vector2.new(vector.X, headVector.Y - 18)
                objs.Text.Visible = true
            else
                objs.Box.Visible = false
                objs.Text.Visible = false
            end
        else
            objs.Box.Visible = false
            objs.Text.Visible = false
        end
    end

    if isAiming then
        local target = getTarget()
        if target then
            camera.CFrame = CFrame.new(camera.CFrame.Position, target.Position)
        end
    end
end)
